#include<Windows.h>
#include"resource1.h"
//状态参数
int g_state = 0;

//回调函数
LRESULT CALLBACK WndProc(HWND hwnd,
    UINT uMsg, 
    WPARAM wParam, 
    LPARAM lParam)
{
	switch (uMsg)
	{
		//菜单消息
	case WM_COMMAND:
		OnCommand(wParam);
		break;
	case WM_KEYDOWN:
		if (GetAsyncKeyState(VK_F1) != 0)
		{
			if (g_state == 0)
			{//开启计时器
				SetTimer(hwnd, 1, 1000, NULL);
				//说明开启
				g_state = 1;
			}
			else
			{//关闭计时器
				KillTimer(hwnd, 1);
				g_state = 0;
				MessageBox(NULL, "成功关闭计时器", "温馨提示", MB_OK);
			}
		}
		break;
	//计时器消息
	case WM_TIMER:
		MessageBox(NULL, "计时器触发", "温馨提示", MB_OK);
		break;
	case WM_LBUTTONDBLCLK:
		MessageBox(NULL, "按下鼠标左键","温馨提示",MB_OK);
		break;
	case WM_CLOSE:
		PostQuitMessage(0);
		break;
	case WM_PAINT:
	{
		PAINTSTRUCT ps = { 0 };
		//获取设备上下文的句柄
		HDC hdc = BeginPaint(hwnd, &ps);
		//绘制文本
		char text[] = "按下F1键!";
		RECT rc = { 0 };
		//获取客户区大小
		GetClientRect(hwnd, &rc);
		//透明标签
		SetBkMode(hdc, TRANSPARENT);
		//绘制文本
		DrawText(hdc, text, -1, &rc, DT_VCENTER| DT_SINGLELINE| DT_CENTER);
		EndPaint(hwnd, &ps);

	}
	default:
		return DefWindowProc(hwnd, uMsg, wParam, lParam);
		break;
	}
	return 0;
}
int APIENTRY wWinMain(_In_ HINSTANCE hInstance,//函数的入口处
                     _In_opt_ HINSTANCE hPrevInstance,
                     _In_ LPWSTR    lpCmdLine,
                     _In_ int       nCmdShow)
{
	MessageBox(NULL, "欢迎使用本地辅助软件", "温馨提示", MB_OK);
	//初始化窗口类结构
    WNDCLASS wnd = {0};
	wnd.style = CS_HREDRAW | CS_VREDRAW;
	wnd.lpfnWndProc = WndProc;
	wnd.hInstance = hInstance;
	wnd.hbrBackground = (HBRUSH)(COLOR_WINDOW + 1);
	wnd.lpszClassName = "MyWindowClass";
	wnd.lpszMenuName = MAKEINTRESOURCE(IDR_MENU1);
	wnd.hIcon = LoadIcon(hInstance, MAKEINTRESOURCE(IDI_ICON1));
	//注册
	RegisterClass(&wnd);
	//创建窗口
	HWND hwnd = CreateWindowA("MyWindowClass", 
		"游戏辅助", 
		WS_OVERLAPPEDWINDOW|WS_VISIBLE, 
		100, 100, 500, 500,
		NULL, 
		NULL, 
		hInstance, NULL);
	//显示和更新窗口
	ShowWindow(hwnd, SW_SHOW);
	UpdateWindow(hwnd);
	//消息循环
	MSG msg = {0};
	while (GetMessage(&msg, NULL, 0, 0))
	{
		
		DispatchMessage(&msg);
	}
	    return 0;
	
}
